🚀 PASOS PARA ACTIVAR (4 PASOS)
═══════════════════════════════════════════════════════════════════════════════

1️⃣  INSTALAR FLUTTERFIRE CLI
    $ dart pub global activate flutterfire_cli

2️⃣  CONFIGURAR FIREBASE
    $ cd solerabravoapp
    $ flutterfire configure
    
    ⏩ Te pedirá seleccionar proyecto y plataformas
    ⏩ Generará google-services.json y GoogleService-Info.plist
    ⏩ Actualizará firebase_options.dart

3️⃣  INSTALAR DEPENDENCIAS
    $ flutter pub get

4️⃣  EJECUTAR
    $ flutter run

  // Verificar si es premium
  if (context.isPremium) {
    // Mostrar funciones premium
  }

  // Proteger contenido
  PremiumOnly(
    child: MyPremiumFeature(),
  )

  // Obtener datos
  String? name = context.userName;
  String? email = context.userEmail;

  // Logout
  context.read<AuthProvider>().logout();
